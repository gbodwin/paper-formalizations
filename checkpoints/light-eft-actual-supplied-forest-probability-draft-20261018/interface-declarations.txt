/-- A property of the actual returned endpoint list, via its typed edge-set
representation. This is a semantic relation, not an uncharged decoder call. -/
def OutputSatisfies (n : ℕ) (P : Finset (Sym2 (Fin n)) → Prop)
    (raw : List StoredEdges.Edge) : Prop :=
  ∃ out : List (Fin n × Fin n),raw=StoredEdges.encode out ∧ P (StoredSample.edgeSet out)

theorem assignment_failure_le {n m : ℕ} {J : Type*} [Fintype J] [DecidableEq J]
    (Q : List (Fin n × Fin n)) (w : Sym2 (Fin n) → ℚ) (hw : ∀ e,0≤w e) (t : ℚ)
    (eligible : Sym2 (Fin n) → Finset J) (host : J → Finset (Sym2 (Fin n))) (h : ℕ)
    (js : Sym2 (Fin n) → List J)
    (input : SampledGreedy.Assignment (Fin n) J m → Sym2 (Fin n) → J → StoredHosts.TypedInput n)
    (es : List (Fin n × Fin n))
    (hj : ∀ e∈es,(js s(e.1,e.2)).Nodup)
    (he : ∀ e∈es,(js s(e.1,e.2)).toFinset=eligible s(e.1,e.2))
    (hh : ∀ a,∀ e∈es,∀ j∈js s(e.1,e.2),host j=StoredSample.edgeSet (input a s(e.1,e.2) j).edges)
    (hd : ∀ a,∀ e∈es,∀ j∈js s(e.1,e.2),
      ((input a s(e.1,e.2) j).draws.map StoredSample.edgeSet)=List.ofFn (a s(e.1,e.2) j))
    (law : FiniteProbability.Law (SampledGreedy.Assignment (Fin n) J m))
    (P : Finset (Sym2 (Fin n)) → Prop) :
    law.prob (fun a => ¬OutputSatisfies n P
      (execute (StoredEdges.encode Q) (StoredEdges.encode Q) (StoredRecords.encodeWeights w)
        n h ((assignmentRecords w t js (input a) es).map encode))) ≤
    law.prob (fun a => ¬P (SampledGreedy.edges (StoredSample.edgeSet Q)
      (fun d => (w d:ℝ)) (t:ℝ) eligible host (SampledGreedy.assignmentTape a) h
      (es.map fun e => s(e.1,e.2)).reverse)) := by
  apply FiniteProbability.Law.prob_mono
  intro a hbad hgood
  obtain ⟨out,run,edges⟩ := execute_assignment Q w hw t eligible host a h js (input a) es hj he (hh a) (hd a)
  exact hbad ⟨out,run,edges.symm ▸ hgood⟩
end LightEFTSpanners.StoredCandidates

noncomputable def componentId {n q : ℕ} (edges : Fin q → List (Fin n × Fin n))
    (j : ForestComponentIndex (StoredForestHosts.family edges)) : ℕ :=
  (StoredForestInput.indexedRecord edges [] j).id
noncomputable def componentVertices {n q : ℕ} (edges : Fin q → List (Fin n × Fin n))
    (j : ForestComponentIndex (StoredForestHosts.family edges)) : List (Fin n) :=
  (StoredForestInput.indexedRecord edges [] j).vertices
noncomputable def hostInput {n q : ℕ} (edges : Fin q → List (Fin n × Fin n))
    (banks : Sym2 (Fin n) → StoredDrawLookup.TypedBank n) (e : Sym2 (Fin n))
    (j : ForestComponentIndex (StoredForestHosts.family edges)) : StoredHosts.TypedInput n :=
  (StoredForestInput.indexedRecord edges (banks e) j).host

def rawInputs {n : ℕ} (banks : Sym2 (Fin n) → StoredDrawLookup.TypedBank n)
    (es : List (Fin n × Fin n)) : List (StoredForestBank.TypedInput n) :=
  es.map fun e => ⟨e,banks s(e.1,e.2)⟩

theorem indexed_prepared {n q : ℕ} (edges : Fin q → List (Fin n × Fin n))
    (banks : Sym2 (Fin n) → StoredDrawLookup.TypedBank n)
    (w : Sym2 (Fin n) → ℚ) (t : ℚ) (es : List (Fin n × Fin n)) :
    StoredForestBank.typedValue (List.ofFn edges) w t (rawInputs banks es)=
      (StoredComponentCandidates.indexedRecords (componentId edges) (componentVertices edges)
        (hostInput edges banks) (StoredForestHosts.indices edges) es).map
        (fun r => StoredCandidatePreparation.typedResult w t (StoredComponentCandidate.typedPrepared r)) := by
  simp only [StoredForestBank.typedValue,rawInputs,StoredComponentCandidates.indexedRecords,List.map_map]
  apply List.map_congr_left
  intro e _
  simp only [Function.comp_apply,StoredForestCandidate.typedResult,StoredForestInput.typedInputs_indexed]
  congr 2

theorem execute_forest_assignment {n m q : ℕ} (edges : Fin (q+1) → List (Fin n × Fin n))
    (Q : List (Fin n × Fin n)) (w : Sym2 (Fin n) → ℚ) (hw : ∀ e,0≤w e) (t : ℚ)
    (a : SampledGreedy.Assignment (Fin n) (ForestComponentIndex (StoredForestHosts.family edges)) m)
    (h : ℕ) (banks : Sym2 (Fin n) → StoredDrawLookup.TypedBank n) (es : List (Fin n × Fin n))
    (hd : ∀ e∈es,∀ j∈StoredForestHosts.indices edges,
      ((hostInput edges banks s(e.1,e.2) j).draws.map StoredSample.edgeSet)=List.ofFn (a s(e.1,e.2) j)) :
    ∃ out : List (Fin n × Fin n),
      execute n h ((List.ofFn edges).map StoredEdges.encode) (StoredEdges.encode Q) (StoredEdges.encode Q)
        (StoredRecords.encodeWeights w) t ((rawInputs banks es).map StoredForestBank.encode)=StoredEdges.encode out ∧
      StoredSample.edgeSet out=SampledGreedy.edges (StoredSample.edgeSet Q) (fun d => (w d:ℝ)) (t:ℝ)
        (SampledGreedy.componentEligible Finset.univ (StoredForestHosts.family edges))
        (SampledGreedy.componentHost (StoredForestHosts.family edges)) (SampledGreedy.assignmentTape a) h
        (es.map fun e => s(e.1,e.2)).reverse := by
  obtain ⟨out,he,ho⟩ := StoredComponentCandidates.execute_forest_assignment Q w hw t
    (StoredForestHosts.family edges) a h (componentId edges) (componentVertices edges)
    (hostInput edges banks) (StoredForestHosts.indices edges) es
    (StoredForestHosts.indices_nodup edges) (StoredForestHosts.indices_univ edges)
    (fun j v => StoredForestInput.indexed_vertices edges [] j v)
    (fun e _ j _ => (StoredForestInput.indexed_host edges (banks s(e.1,e.2)) j).symm) hd
  refine ⟨out,?_,ho⟩
  rw [execute_encoded,indexed_prepared]
  rw [StoredComponentCandidates.execute_encoded] at he
  exact he
end LightEFTSpanners.StoredForestCandidates

def rawHostBound (n q m : ℕ) (fs : List (List StoredEdges.Edge)) (bank : StoredDrawLookup.Bank) : ℕ :=
  (bank.map fun r => r.2.length).sum*
    (n*n*(q+m+(fs.map List.length).sum+11)+4*n+SampleVote.perSample n+5)+
    n*n*(bank.map fun r => StoredVote.drawCells r.2).sum+4

def selectionBound (n q m : ℕ) (fs : List (List StoredEdges.Edge)) (rs : List StoredForestBank.Input) : ℕ :=
  (rs.map fun r => fs.length*n*(rawHostBound n q m fs r.bank+2)+3).sum+1

def bound (n : ℕ) (fs : List (List StoredEdges.Edge)) (Q E : List StoredEdges.Edge)
    (rs : List StoredForestBank.Input) : ℕ :=
  StoredForestBank.bound n fs rs+1+selectionBound n Q.length (E.length+rs.length) fs rs

def execute (n h : ℕ) (fs : List (List StoredEdges.Edge)) (Q E : List StoredEdges.Edge)
    (W : StoredRecords.Weights) (t : ℚ) (rs : List StoredForestBank.Input) : List StoredEdges.Edge :=
  match Controller.run (transition fs Q E W n h t) (bound n fs Q E rs) (.prepare (.inputs rs [])) with
  | .select (.halt out) => out
  | _ => E

theorem raw_host_encoded {n : ℕ} (q m : ℕ) (fs : List (List (Fin n × Fin n)))
    (bank : StoredDrawLookup.TypedBank n) :
    rawHostBound n q m (fs.map StoredEdges.encode) (StoredDrawLookup.encode bank)=
      StoredForestBank.hostBound q m fs bank := by
  simp [rawHostBound,StoredForestBank.hostBound,StoredForestBank.drawCount,
    StoredForestBank.edgeCount,StoredForestBank.drawCells,StoredDrawLookup.encode,
    StoredVote.encodeDraws,StoredVote.drawCells,StoredEdges.encode,List.map_map,Function.comp_def]

theorem work_le_bound {n : ℕ} (fs : List (List (Fin n × Fin n))) (Q E : List (Fin n × Fin n))
    (w : Sym2 (Fin n) → ℚ) (h : ℕ) (t : ℚ) (rs : List (StoredForestBank.TypedInput n)) :
    StoredCandidates.work Q w h (StoredForestBank.typedValue fs w t rs) E ≤
      selectionBound n Q.length (E.length+rs.length) (fs.map StoredEdges.encode) (rs.map StoredForestBank.encode) := by
  have hw := StoredCandidates.work_le_input Q E w h (E.length+rs.length)
    (StoredForestBank.typedValue fs w t rs) (by simp [StoredForestBank.typedValue])
  apply hw.trans
  simp only [StoredForestBank.typedValue,selectionBound,List.map_map,Function.comp_def,
    StoredForestBank.encode,List.length_map,raw_host_encoded]
  apply Nat.add_le_add_right
  clear hw
  generalize E.length+rs.length = m
  induction rs with
  | nil => exact Nat.le_refl _
  | cons r rs ih =>
    have hc := StoredForestBank.candidate_cap_le Q.length m fs w t r
    simp only [List.map_cons,List.sum_cons]
    exact Nat.add_le_add hc ih

theorem full_trace {n : ℕ} (fs : List (List (Fin n × Fin n))) (Q E : List (Fin n × Fin n))
    (w : Sym2 (Fin n) → ℚ) (h : ℕ) (t : ℚ) (rs : List (StoredForestBank.TypedInput n)) :
    ∃ k, k ≤ bound n (fs.map StoredEdges.encode) (StoredEdges.encode Q) (StoredEdges.encode E)
      (rs.map StoredForestBank.encode) ∧
      Trace (transition (fs.map StoredEdges.encode) (StoredEdges.encode Q) (StoredEdges.encode E)
        (StoredRecords.encodeWeights w) n h t) k (.prepare (.inputs (rs.map StoredForestBank.encode) []))
        (.select (.halt (StoredEdges.encode (StoredCandidates.selected Q w h (StoredForestBank.typedValue fs w t rs) E)))) := by
  obtain ⟨k,hk,ht⟩ := prepared_trace fs Q E w h t rs
  have hb := work_le_bound fs Q E w h t rs
  exact ⟨k,by simp only [bound,StoredEdges.encode,List.length_map];omega,ht⟩

theorem execute_encoded {n : ℕ} (fs : List (List (Fin n × Fin n))) (Q E : List (Fin n × Fin n))
    (w : Sym2 (Fin n) → ℚ) (h : ℕ) (t : ℚ) (rs : List (StoredForestBank.TypedInput n)) :
    execute n h (fs.map StoredEdges.encode) (StoredEdges.encode Q) (StoredEdges.encode E)
      (StoredRecords.encodeWeights w) t (rs.map StoredForestBank.encode)=
      StoredEdges.encode (StoredCandidates.selected Q w h (StoredForestBank.typedValue fs w t rs) E) := by
  obtain ⟨k,hk,ht⟩ := full_trace fs Q E w h t rs
  have he : bound n (fs.map StoredEdges.encode) (StoredEdges.encode Q) (StoredEdges.encode E)
      (rs.map StoredForestBank.encode)=k+(bound n (fs.map StoredEdges.encode)
        (StoredEdges.encode Q) (StoredEdges.encode E) (rs.map StoredForestBank.encode)-k) := by omega
  unfold execute
  rw [he,Controller.run_add,ht.run_eq,Controller.run_terminal _ _ rfl]
end LightEFTSpanners.StoredForestCandidates

def family {n q : ℕ} (edges : Fin q → List (Fin n × Fin n)) : Fin q → SimpleGraph (Fin n) :=
  fun i => ExecutableVoting.graph (StoredSample.edgeSet (edges i))

def indices {n q : ℕ} (edges : Fin q → List (Fin n × Fin n)) : List (ForestComponentIndex (family edges)) :=
  (List.finRange q).flatMap fun i => (StoredCanonicalComponents.indices (edges i)).map (Sigma.mk i)

theorem indices_all {n q : ℕ} (edges : Fin q → List (Fin n × Fin n))
    (j : ForestComponentIndex (family edges)) : j∈indices edges := by
  obtain ⟨i,c⟩ := j
  exact List.mem_flatMap.mpr ⟨i,by simp,List.mem_map.mpr ⟨c,StoredCanonicalComponents.indices_all (edges i) c,rfl⟩⟩

theorem indices_nodup {n q : ℕ} (edges : Fin q → List (Fin n × Fin n)) : (indices edges).Nodup := by
  apply List.nodup_flatMap.mpr
  constructor
  · intro i _
    exact (StoredCanonicalComponents.indices_nodup (edges i)).map (fun _ _ h => by cases h; rfl)
  · apply (List.nodup_finRange q).imp
    intro i j hij
    apply List.disjoint_left.mpr
    intro x hx hy
    obtain ⟨c,_,hc⟩ := List.mem_map.mp hx
    obtain ⟨d,_,hd⟩ := List.mem_map.mp hy
    exact hij (congrArg Sigma.fst (hc.trans hd.symm))

theorem indices_univ {n q : ℕ} (edges : Fin q → List (Fin n × Fin n)) :
    (indices edges).toFinset=Finset.univ := by
  classical
  exact Finset.eq_univ_of_forall fun j => List.mem_toFinset.mpr (indices_all edges j)

def label (n : ℕ) (i r : ℕ) : ℕ := i*n+r

theorem label_injective {n i j r s : ℕ} (hr : r<n) (hs : s<n)
    (h : label n i r=label n j s) : i=j ∧ r=s := by
  have hn : 0<n := by omega
  have hi : label n i r/n=i := by
    simpa [label,Nat.add_comm,Nat.div_eq_of_lt hr] using Nat.add_mul_div_right r i hn
  have hj : label n j s/n=j := by
    simpa [label,Nat.add_comm,Nat.div_eq_of_lt hs] using Nat.add_mul_div_right s j hn
  have he : i=j := hi.symm.trans ((congrArg (fun x => x/n) h).trans hj)
  subst j
  exact ⟨rfl,by simpa [label] using h⟩
end LightEFTSpanners.StoredForestHosts

/-- The empirical-vote output on the actual weight-sorted non-seed edge list. -/
noncomputable def output (G Q : SimpleGraph V) (w : Sym2 V → ℝ) (t : ℝ)
    (eligible : Sym2 V → Finset J) (host : J → Finset (Sym2 V)) (tape : Tape V J) (h : ℕ) :
    SimpleGraph V := edgeGraph (edges Q.edgeFinset w t eligible host tape h (input G Q w))

 theorem output_isEFTSpanner (G Q : SimpleGraph V) (hQ : Q≤G)
    (w : Sym2 V → ℝ) (t : ℝ) (f h : ℕ)
    (eligible : Sym2 V → Finset J) (host : J → Finset (Sym2 V)) (tape : Tape V J)
    (hf : 0<f) (ht : 1≤t) (hw : ∀ e,0≤w e)
    (hhost : ∀ e∈G.edgeFinset\Q.edgeFinset,∀ j∈eligible e,host j⊆Q.edgeFinset)
    (hcount : ∀ e∈G.edgeFinset\Q.edgeFinset,2*f+h≤(eligible e).card)
    (hcong : ∀ e∈G.edgeFinset\Q.edgeFinset,∀ d,(hosts (eligible e) host d).card≤2)
    (hacc : AccurateTrace Q.edgeFinset w t f eligible host tape h (input G Q w)) :
    IsEFTSpanner G (output G Q w t eligible host tape h) w t f := by
  have hgraph : edgeGraph (Q.edgeFinset∪(input G Q w).toFinset)=G := by
    apply SimpleGraph.edgeSet_injective
    ext e
    simp only [mem_edgeGraph,mem_union,List.mem_toFinset,mem_input,mem_edgeFinset]
    constructor
    · rintro ⟨he,_⟩
      rcases he with he | ⟨he,_⟩
      · exact SimpleGraph.edgeSet_mono hQ he
      · exact he
    · intro he
      refine ⟨?_,G.not_isDiag_of_mem_edgeSet he⟩
      by_cases heQ : e∈Q.edgeSet
      · exact Or.inl heQ
      · exact Or.inr ⟨he,heQ⟩
  have hin : ∀ e∈input G Q w,e∈G.edgeFinset\Q.edgeFinset := by
    intro e he
    exact mem_sdiff.mpr ((mem_input G Q w e).mp he)
  have h := edges_isEFTSpanner Q.edgeFinset w t f h eligible host tape hf ht hw
    (input G Q w) (fun e he => G.not_isDiag_of_mem_edgeFinset (mem_sdiff.mp (hin e he)).1)
    (fun e he => hhost e (hin e he)) (fun e he => hcount e (hin e he))
    (fun e he => hcong e (hin e he)) hacc
  simpa only [hgraph,output] using h
end LightEFTSpanners.SampledGreedy

/-- Failure of the full supplied-forest package is contained in trace
inaccuracy for the very same finite fresh-assignment law. -/
theorem polynomial_supplied_forest_package_failure_probability
    (G Q : SimpleGraph V) (q f h k s : ℕ) (w : Sym2 V → ℝ) (eps : ℝ)
    (hQ : IsFTConnectivityPreserver G Q q)
    (F : Fin (q+1) → SimpleGraph V)
    (hF : ∀ i, F i ≤ Q ∧ (F i).IsAcyclic)
    (hcount : ∀ e ∈ G.edgeFinset \ Q.edgeFinset,
      2*f+h ≤ (componentEligible univ F e).card)
    (hcong : ∀ d : Sym2 V,
      (hosts (univ : Finset (Fin (q+1))) (fun i => (F i).edgeFinset) d).card ≤ 2)
    (hf : 0 < f) (hh : 0 < h) (hk : 0 < k) (heps : 0 < eps)
    (hw : ∀ e, 0 ≤ w e) (hwpos : ∀ e ∈ G.edgeSet, 0 < w e) :
    let t : ℝ := (1+eps)*(2*k-1)
    let B : ℝ := 1+64*(f:ℝ)*extremalLightness (Fintype.card V) (t+1)/h
    (executionLaw (V:=V) (J:=ForestComponentIndex F) f hf
      (polynomialRepetitions (Fintype.card V) s) (input G Q w)).prob (fun a =>
        let H := output G Q w t (componentEligible univ F) (componentHost F)
          (assignmentTape a) h
        ¬(IsEFTSpanner G H w t f ∧
          totalWeight H w ≤ B * totalWeight Q w ∧
          (totalWeight Q w = 0 → totalWeight H w = 0) ∧
          competitiveLightness H Q w ≤ B)) ≤ ((Fintype.card V:ℝ)^s)⁻¹ := by
  classical
  let t : ℝ := (1+eps)*(2*k-1)
  let B : ℝ := 1+64*(f:ℝ)*extremalLightness (Fintype.card V) (t+1)/h
  let μ := executionLaw (V:=V) (J:=ForestComponentIndex F) f hf
    (polynomialRepetitions (Fintype.card V) s) (input G Q w)
  let H := fun a : Assignment V (ForestComponentIndex F)
      (polynomialRepetitions (Fintype.card V) s) =>
    output G Q w t (componentEligible univ F) (componentHost F) (assignmentTape a) h
  change μ.prob (fun a => ¬(IsEFTSpanner G (H a) w t f ∧
    totalWeight (H a) w ≤ B * totalWeight Q w ∧
    (totalWeight Q w = 0 → totalWeight (H a) w = 0) ∧
    competitiveLightness (H a) Q w ≤ B)) ≤ ((Fintype.card V:ℝ)^s)⁻¹
  have hmono :
      μ.prob (fun a => ¬(IsEFTSpanner G (H a) w t f ∧
        totalWeight (H a) w ≤ B * totalWeight Q w ∧
        (totalWeight Q w = 0 → totalWeight (H a) w = 0) ∧
        competitiveLightness (H a) Q w ≤ B)) ≤
      μ.prob (fun a => ¬AccurateTrace Q.edgeFinset w t f
        (componentEligible univ F) (componentHost F) (assignmentTape a) h (input G Q w)) := by
    apply Law.prob_mono
    intro a hbad hacc
    have hgood := voting_good_event_of_supplied_forests G Q hQ.1 q f h k w eps

event. The direct eligible-host count replaces any separate fault-budget
inequality. The raw bound and its zero-weight consequence are retained. -/
theorem voting_good_event_of_supplied_forests
    (G Q : SimpleGraph V) (hQG : Q ≤ G)
    (q f h k : ℕ) (w : Sym2 V → ℝ) (eps : ℝ)
    (F : Fin (q+1) → SimpleGraph V)
    (hF : ∀ i, F i ≤ Q ∧ (F i).IsAcyclic)
    (hcount : ∀ e ∈ G.edgeFinset \ Q.edgeFinset,
      2*f+h ≤ (componentEligible univ F e).card)
    (hcong : ∀ d : Sym2 V,
      (hosts (univ : Finset (Fin (q+1))) (fun i => (F i).edgeFinset) d).card ≤ 2)
    (hf : 0 < f) (hh : 0 < h) (hk : 0 < k) (heps : 0 < eps)
    (hw : ∀ e, 0 ≤ w e) (hwpos : ∀ e ∈ G.edgeSet, 0 < w e)
    (tape : Tape V (ForestComponentIndex F))
    (hacc : AccurateTrace Q.edgeFinset w ((1+eps)*(2*k-1)) f
      (componentEligible univ F) (componentHost F) tape h (input G Q w)) :
    let t : ℝ := (1+eps)*(2*k-1)
    let H := output G Q w t (componentEligible univ F) (componentHost F) tape h
    let B : ℝ := 1+64*(f:ℝ)*extremalLightness (Fintype.card V) (t+1)/h
    IsEFTSpanner G H w t f ∧
      totalWeight H w ≤ B * totalWeight Q w ∧
      (totalWeight Q w = 0 → totalWeight H w = 0) ∧
      competitiveLightness H Q w ≤ B := by
  classical
  let t : ℝ := (1+eps)*(2*k-1)
  let H := output G Q w t (componentEligible univ F) (componentHost F) tape h
  let B : ℝ := 1+64*(f:ℝ)*extremalLightness (Fintype.card V) (t+1)/h
  change IsEFTSpanner G H w t f ∧
    totalWeight H w ≤ B * totalWeight Q w ∧
    (totalWeight Q w = 0 → totalWeight H w = 0) ∧
    competitiveLightness H Q w ≤ B
  have ht : 1 ≤ t := by
    have hthreshold := actual_threshold_gt_two heps hk
    dsimp only [t]
    linarith
  have hfull (d : Sym2 V) :
      (hosts (forestComponentIndices univ F) (componentHost F) d).card ≤ 2 :=
    (forest_component_congestion univ F d).trans (hcong d)
  have hlocal (e d : Sym2 V) :
      (hosts (componentEligible univ F e) (componentHost F) d).card ≤ 2 := by
    apply le_trans (card_le_card ?_) (hfull d)
    intro p hp
    obtain ⟨hp,hd⟩ := mem_filter.mp hp
    exact mem_filter.mpr ⟨(mem_filter.mp hp).1,hd⟩
  have hhost (p : ForestComponentIndex F) : componentHost F p ⊆ Q.edgeFinset :=
    edgeFinset_mono ((component_tree_le (F p.1) p.2).trans (hF p.1).1)
  have hcorrect : IsEFTSpanner G H w t f := by
    exact output_isEFTSpanner G Q hQG w t f h
      (componentEligible univ F) (componentHost F) tape hf ht hw
      (fun _ _ p _ => hhost p) hcount (fun e _ d => hlocal e d) hacc
  have hweight : totalWeight H w ≤ B * totalWeight Q w := by
    let A : ForestComponentIndex F → Type _ := fun p => p.2
    let j : ∀ p, A p ↪ V := fun p => componentEmbedding (F p.1) p.2
    let T : ∀ p, SimpleGraph (A p) := fun p => p.2.toSimpleGraph
    apply voting_output_weight_bound G Q hQG w eps k f h hk heps hf hh
      (forestComponentIndices univ F) j T (componentEligible univ F) tape
      (fun p _ => component_isTree (hF p.1).2 p.2)
      (fun p _ => (component_tree_le (F p.1) p.2).trans (hF p.1).1)
      (fun _ _ => filter_subset _ _) (fun _ _ p hp => (mem_filter.mp hp).2)
      (fun d _ => hfull d) hwpos hacc
  have hzero : totalWeight Q w = 0 → totalWeight H w = 0 := by
    intro hz
    have hnonneg : 0 ≤ totalWeight H w := by
      unfold totalWeight
      exact sum_nonneg (fun e _ => hw e)
    have hupper : totalWeight H w ≤ 0 := by
      simpa only [hz,mul_zero] using hweight
    exact le_antisymm hupper hnonneg
  have hB : 0 ≤ B := by
    have hLambda := extremalLightness_nonneg (Fintype.card V)
      (actual_threshold_gt_two heps hk)
    dsimp only [B,t]
    positivity
  have hratio : competitiveLightness H Q w ≤ B := by
    by_cases hz : totalWeight Q w = 0
    · have houtzero : totalWeight H w = 0 := hzero hz
      simpa only [competitiveLightness,hz,houtzero,div_zero] using hB
    · have hQnonneg : 0 ≤ totalWeight Q w := by
        unfold totalWeight
        exact sum_nonneg (fun e _ => hw e)
      have hQpos : 0 < totalWeight Q w := lt_of_le_of_ne hQnonneg (Ne.symm hz)
      exact (div_le_iff₀ hQpos).mpr hweight
  exact ⟨hcorrect,hweight,hzero,hratio⟩
