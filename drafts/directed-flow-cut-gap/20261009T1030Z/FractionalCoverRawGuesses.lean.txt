import DirectedFlowCutGap.FractionalCoverRawGraph
import DirectedFlowCutGap.FractionalCoverGuesses

/-!
# Original-input raw dispatch and lambda list

Zero tests inspect stored natural numerators. All shortest paths, minimum scans,
lambda values, shifted costs and cover solves use unreduced natural fractions.
The whole returned data result refines the frozen rational dispatch/guess code;
no rational denotation is evaluated by this executable entry point.
-/
namespace DirectedFlowCutGap.FractionalCoverRawGuesses
open scoped BigOperators
open FractionalCover FractionalCoverRawCore RawNonnegativeRational IntegralNetworkFlow
open FractionalCoverRawGraph
variable {n : ℕ}

@[simp] lemma rational_eq_zero (q : Code) : rational q=0 ↔ q.num=0 := by
  simpa only [rational,NNRat.coe_eq_zero] using Code.value_eq_zero q

@[simp] lemma rational_pos (q : Code) : 0<rational q ↔ 0<q.num := by
  exact_mod_cast Code.value_pos q

def zeroSet (c : RawRow n) : Column n := Finset.univ.filter (fun i => (get c i).num=0)

lemma zeroSet_refines (c : RawRow n) : zeroSet c = FractionalCoverDispatch.zeroSet (decodeRow c) := by
  ext i
  simp [zeroSet,FractionalCoverDispatch.zeroSet]

def findAvoiding (G : Digraph (Fin n)) [DecidableRel G.Adj] (X : Column n) :
    List (Pair n) → Option (Selected n)
  | [] => none
  | d::ds =>
    match (FractionalCoverRawOracle.shortest (ResidualSearch.Enumeration.fin n)
        (FractionalCoverDispatch.keptGraph G X d.1 d.2) (fun _ => Code.zero) d.1 d.2).1 with
    | none => findAvoiding G X ds
    | some p => some ⟨d,p⟩

lemma findAvoiding_refines (G : Digraph (Fin n)) [DecidableRel G.Adj] (X : Column n)
    (ds : List (Pair n)) : (findAvoiding G X ds).map decodeSelected =
      FractionalCoverDispatch.findAvoiding G X ds := by
  induction ds with
  | nil => rfl
  | cons d ds ih =>
    have h := congrArg Prod.fst (FractionalCoverRawOracle.shortest_refines
      (ResidualSearch.Enumeration.fin n) (FractionalCoverDispatch.keptGraph G X d.1 d.2)
      (fun _ => Code.zero) d.1 d.2)
    dsimp only [FractionalCoverRawOracle.decodeResult] at h
    simp only [rational_zero] at h
    cases hr : (FractionalCoverRawOracle.shortest (ResidualSearch.Enumeration.fin n)
        (FractionalCoverDispatch.keptGraph G X d.1 d.2) (fun _ => Code.zero) d.1 d.2).1 <;>
      simp [findAvoiding,FractionalCoverDispatch.findAvoiding,← h,hr,ih,decodeSelected]

inductive DispatchResult (n : ℕ) where
  | infeasible (demand : Pair n)
  | zero (cut : Column n)
  | positive (witness : Selected n)

def decodeDispatch : DispatchResult n → FractionalCoverDispatch.Result n
  | .infeasible d => .infeasible d
  | .zero X => .zero X
  | .positive q => .positive (decodeSelected q)

def dispatch (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) : DispatchResult n :=
  match FractionalCoverDispatch.emptyDemand G ds with
  | some d => .infeasible d
  | none => match findAvoiding G (zeroSet c) ds with
    | none => .zero (zeroSet c)
    | some q => .positive q

lemma dispatch_refines (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) : decodeDispatch (dispatch G ds c) =
      FractionalCoverDispatch.dispatch G ds (decodeRow c) := by
  have h := findAvoiding_refines G (zeroSet c) ds
  unfold dispatch FractionalCoverDispatch.dispatch
  rw [← zeroSet_refines,← h]
  cases FractionalCoverDispatch.emptyDemand G ds with
  | some d => rfl
  | none =>
    cases findAvoiding G (zeroSet c) ds <;> rfl

def minimumPositive (c : RawRow n) : Option (Fin n) :=
  argminCode (get c) ((List.finRange n).filter (fun i => decide (0 < (get c i).num)))

lemma minimumPositive_refines (c : RawRow n) : minimumPositive c =
    FractionalCoverDispatch.minimumPositive (decodeRow c) := by
  simp only [minimumPositive,argminCode_refines,FractionalCoverDispatch.minimumPositive,
    decode_value,rational_pos]

def lambdaCode (c : RawRow n) (j : Fin n) (k : ℕ) : Code :=
  ((get c j).div (Code.ofNat n)).mul (Code.ofNat (2^k))

def shiftedCode (c : RawRow n) (lambda : Code) : RawRow n :=
  Vector.ofFn (fun i => (get c i).add lambda)

@[simp] lemma lambdaCode_value (c : RawRow n) (j : Fin n) (k : ℕ) :
    rational (lambdaCode c j k) = FractionalCoverGuesses.lambda (decodeRow c) j k := by
  simp [lambdaCode,FractionalCoverGuesses.lambda]

@[simp] lemma shiftedCode_value (c : RawRow n) (lambda : Code) :
    decodeRow (shiftedCode c lambda) = FractionalCoverGuesses.shifted (decodeRow c) (rational lambda) := by
  apply Vector.ext
  intro i hi
  simp [shiftedCode,FractionalCoverGuesses.shifted,decodeRow]

def guess (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (j : Fin n) (k : ℕ) : RawState n :=
  FractionalCoverRawGraph.solveGraph G ds (shiftedCode c (lambdaCode c j k)) (Nat.zero_lt_of_lt j.isLt)

lemma guess_refines (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (j : Fin n) (k : ℕ) : decodeState (guess G ds c j k) =
      FractionalCoverGuesses.guess G ds (decodeRow c) j k := by
  unfold guess FractionalCoverGuesses.guess
  rw [FractionalCoverRawGraph.solveGraph_refines,shiftedCode_value,lambdaCode_value]

def guesses (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (j : Fin n) (B : ℕ) : List (RawState n) :=
  (List.range (FractionalCoverGuesses.gridSteps n B+1)).map (guess G ds c j)

lemma guesses_refines (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (j : Fin n) (B : ℕ) : (guesses G ds c j B).map decodeState =
      FractionalCoverGuesses.guesses G ds (decodeRow c) j B := by
  simp [guesses,FractionalCoverGuesses.guesses,List.map_map,Function.comp_def,guess_refines]

inductive Output (n : ℕ) where
  | infeasible (demand : Pair n)
  | zero (cut : Column n)
  | covers (states : List (RawState n))
  | invalid

def decodeOutput : Output n → FractionalCoverGuesses.Output n
  | .infeasible d => .infeasible d
  | .zero X => .zero X
  | .covers states => .covers (states.map decodeState)
  | .invalid => .invalid

def run (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (B : ℕ) : Output n :=
  match dispatch G ds c with
  | .infeasible d => .infeasible d
  | .zero X => .zero X
  | .positive _ => match minimumPositive c with
    | none => .invalid
    | some j => .covers (guesses G ds c j B)

theorem run_refines (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (B : ℕ) : decodeOutput (run G ds c B) =
      FractionalCoverGuesses.run G ds (decodeRow c) B := by
  unfold run FractionalCoverGuesses.run
  rw [← dispatch_refines,← minimumPositive_refines]
  cases dispatch G ds c with
  | infeasible d => rfl
  | zero X => rfl
  | positive q =>
    cases minimumPositive c with
    | none => rfl
    | some j => simp only [decodeDispatch,decodeOutput,guesses_refines]

/-- Full original-input correctness transfers from exact data refinement.
Only actual raw input magnitudes are premises; raw nonnegativity is automatic. -/
theorem run_correct (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (B : ℕ) (hc : RowBounded c B) :
    match decodeOutput (run G ds c B) with
    | .infeasible _ => ¬∃ w : Fin n → ℝ, RealFeasible (FractionalCoverGraphOracle.columns G ds) w
    | .zero X => IsIntegralCut G X {d | d ∈ ds} ∧ (∑ i ∈ X, value (decodeRow c) i)=0
    | .covers states => states.length=FractionalCoverGuesses.gridSteps n B+1 ∧
        (∀ out ∈ states, Feasible (FractionalCoverGraphOracle.columns G ds) out.best) ∧
        ∀ w : Fin n → ℝ, RealFeasible (FractionalCoverGraphOracle.columns G ds) w →
          ∃ out ∈ states, Feasible (FractionalCoverGraphOracle.columns G ds) out.best ∧
            (objective (decodeRow c) out.best : ℝ) ≤ 6*∑ i,(value (decodeRow c) i : ℝ)*w i ∧
            (∑ i,(value out.best i : ℝ)) ≤ 12*∑ i,w i
    | .invalid => False := by
  rw [run_refines]
  have hb := decoded_input_bounds hc
  exact FractionalCoverGuesses.run_correct G ds (decodeRow c)
    (fun i => rational_nonneg (get c i)) B hb.1 hb.2

lemma lambdaCode_bounded {c : RawRow n} {B : ℕ} (hc : RowBounded c B)
    (j : Fin n) (k : ℕ) : (lambdaCode c j k).Bounded (B+Nat.size n+k) := by
  exact Code.bounded_mul (Code.bounded_div (hc j)
    (ofNat_bounded n (Nat.size n) (Nat.lt_size_self n).le))
    (ofNat_bounded (2^k) k le_rfl)

lemma shiftedCode_bounded {c : RawRow n} {B b : ℕ} (hc : RowBounded c B)
    {lambda : Code} (hl : lambda.Bounded b) : RowBounded (shiftedCode c lambda) (B+b+1) := by
  intro i
  exact Code.bounded_add (hc i) hl

def guessInputWidth (n B : ℕ) : ℕ := 2*B+Nat.size n+FractionalCoverGuesses.gridSteps n B+1

lemma guess_input_bounded {c : RawRow n} {B : ℕ} (hc : RowBounded c B)
    (j : Fin n) (k : ℕ) (hk : k ≤ FractionalCoverGuesses.gridSteps n B) :
    RowBounded (shiftedCode c (lambdaCode c j k)) (guessInputWidth n B) := by
  exact rowBounded_mono (shiftedCode_bounded hc (lambdaCode_bounded hc j k))
    (by unfold guessInputWidth; omega)

/-- Every actual raw guess has a stored output-width bound derived solely from
n and the input Code bound B, including its raw lambda and shifted costs. -/
theorem guesses_stored_bits (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    {c : RawRow n} {B : ℕ} (hc : RowBounded c B) (j : Fin n) :
    ∀ out ∈ guesses G ds c j B, ∀ i : Fin n,
      Nat.size (get out.best i).num ≤
        normalWidth n (weightWidth n (guessInputWidth n B) (fuel n))+1 ∧
      Nat.size (get out.best i).den ≤
        normalWidth n (weightWidth n (guessInputWidth n B) (fuel n))+1 := by
  intro out ho i
  obtain ⟨k,hk,rfl⟩ := List.mem_map.mp ho
  have hkle : k ≤ FractionalCoverGuesses.gridSteps n B := by
    have := List.mem_range.mp hk
    omega
  exact solve_stored_bits (guess_input_bounded hc j k hkle) _ i

/-- Exact event-based oracle-call total across the actual raw guess family. -/
theorem guesses_oracle_calls (G : Digraph (Fin n)) [DecidableRel G.Adj] (ds : List (Pair n))
    (c : RawRow n) (j : Fin n) (B : ℕ) :
    ((guesses G ds c j B).map FractionalCoverRawCore.oracleCalls).sum ≤
      (FractionalCoverGuesses.gridSteps n B+1)*(3*n^2+1) := by
  have hlist (ks : List ℕ) :
      ((ks.map (guess G ds c j)).map FractionalCoverRawCore.oracleCalls).sum ≤
        ks.length*(3*n^2+1) := by
    induction ks with
    | nil => simp
    | cons k ks ih =>
      have h := FractionalCoverRawCore.solve_oracle_calls
        (shiftedCode c (lambdaCode c j k))
        (FractionalCoverRawGraph.oracle G ds (shiftedCode c (lambdaCode c j k))
          (Nat.zero_lt_of_lt j.isLt))
      change FractionalCoverRawCore.oracleCalls (guess G ds c j k) ≤ 3*n^2+1 at h
      simp only [List.map_cons,List.sum_cons,List.length_cons]
      nlinarith
  simpa only [guesses,List.length_range] using
    hlist (List.range (FractionalCoverGuesses.gridSteps n B+1))

/-- This is the complete raw finite-data dispatch/guess entry point. -/
def runInput (D : FractionalCoverRawGraph.Input n) (B : ℕ) : Output n :=
  run D.graph D.demands D.costs B

/-- Derive the grid horizon from the stored input itself. No width certificate
or numerical bound is supplied by the caller of this finite-data function. -/
def solveInput (D : FractionalCoverRawGraph.Input n) : Output n :=
  runInput D (inputWidth D.costs)

theorem solveInput_refines (D : FractionalCoverRawGraph.Input n) :
    decodeOutput (solveInput D) = FractionalCoverGuesses.run D.graph D.demands
      (decodeRow D.costs) (inputWidth D.costs) :=
  run_refines D.graph D.demands D.costs (inputWidth D.costs)

theorem solveInput_not_invalid (D : FractionalCoverRawGraph.Input n) :
    solveInput D ≠ .invalid := by
  intro h
  have hs := run_correct D.graph D.demands D.costs (inputWidth D.costs) (inputWidth_bound D.costs)
  change solveInput D = .invalid at h
  change (match decodeOutput (solveInput D) with
    | .infeasible _ => _
    | .zero _ => _
    | .covers _ => _
    | .invalid => False) at hs
  rw [h] at hs
  exact hs

end DirectedFlowCutGap.FractionalCoverRawGuesses
