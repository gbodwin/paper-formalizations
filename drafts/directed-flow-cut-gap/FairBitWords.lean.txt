import DirectedFlowCutGap.FiniteGridSampler
import DirectedFlowCutGap.FiniteAmplification
import Mathlib.Data.Nat.Log

/-!
# Explicit fair-bit words and one rejection trial

A bit is a uniform element of `Fin 2`. The executable interpreter uses one
doubling/addition per bit. It decodes independent bits bijectively into a
bounded binary integer. A rejection trial keeps exactly the integers below the
requested bound. Its rejection probability is at most one half.

This module does not identify a truncated rejection sampler with an exact
uniform draw and does not yet compose a random-bit implementation with the
adaptive graph algorithm.
-/
namespace DirectedFlowCutGap.FairBitWords
open scoped BigOperators ENNReal
open FinitePermutationSampler FiniteAmplification

abbrev Bits (b : ℕ) := FiniteGridSampler.Cells 2 b

/-- Binary positional decoding, with the current bit as the low-order bit. -/
def decode : (b : ℕ) → Bits b ≃ Fin (2 ^ b)
  | 0 => Equiv.ofUnique Unit (Fin 1)
  | b + 1 =>
      ((Equiv.prodCongr (Equiv.refl (Fin 2)) (decode b)).trans
        (Equiv.prodComm (Fin 2) (Fin (2 ^ b)))).trans
          (finProdFinEquiv.trans (finCongr (pow_succ 2 b).symm))

/-- The number and the bit-read counter are returned by the same recursion. -/
def run : (b : ℕ) → Bits b → ℕ × ℕ
  | 0, _ => (0, 0)
  | b + 1, (j, t) =>
      let r := run b t
      (2 * r.1 + j.val, r.2 + 1)

theorem run_value (b : ℕ) (t : Bits b) : (run b t).1 = (decode b t).val := by
  induction b with
  | zero => rfl
  | succ b ih =>
      rcases t with ⟨j,t⟩
      simp only [run, decode, Equiv.trans_apply, Equiv.prodCongr_apply,
        Equiv.refl_apply, Equiv.prodComm_apply, finProdFinEquiv_apply,
        Fin.coe_cast, ih]
      omega

@[simp] theorem run_bits (b : ℕ) (t : Bits b) : (run b t).2 = b := by
  induction b with
  | zero => rfl
  | succ b ih => rcases t with ⟨j,t⟩; simp [run, ih]

theorem run_value_lt (b : ℕ) (t : Bits b) : (run b t).1 < 2 ^ b := by
  rw [run_value]
  exact (decode b t).isLt

/-- This counts the two literal arithmetic operations in each recursive arm. -/
def arithmeticOps : (b : ℕ) → Bits b → ℕ
  | 0, _ => 0
  | b + 1, (_,t) => arithmeticOps b t + 2

@[simp] theorem arithmeticOps_eq (b : ℕ) (t : Bits b) : arithmeticOps b t = 2 * b := by
  induction b with
  | zero => rfl
  | succ b ih => rcases t with ⟨j,t⟩; simp [arithmeticOps, ih]; omega

def width (n : ℕ) : ℕ := Nat.log2 n + 1
def bucket (n : ℕ) : ℕ := 2 ^ width n

theorem width_pos (n : ℕ) : 0 < width n := by unfold width; omega
theorem bucket_pos (n : ℕ) : 0 < bucket n := by unfold bucket; positivity

theorem lt_bucket (n : ℕ) : n < bucket n := by
  simpa only [bucket, width, Nat.log2_eq_log_two] using
    Nat.lt_pow_succ_log_self (by decide : 1 < 2) n

theorem bucket_le_twice {n : ℕ} (hn : 0 < n) : bucket n ≤ 2 * n := by
  have h := Nat.pow_log_le_self 2 (by omega : n ≠ 0)
  simpa only [bucket, width, Nat.log2_eq_log_two, pow_succ, Nat.mul_comm] using
    Nat.mul_le_mul_left 2 h

/-- A trial either returns the legal requested index or reports rejection. -/
def accept (n : ℕ) (x : Fin (bucket n)) : Option (Fin n) :=
  if h : x.val < n then some ⟨x.val,h⟩ else none

def embed {n : ℕ} (a : Fin n) : Fin (bucket n) :=
  ⟨a.val, a.isLt.trans (lt_bucket n)⟩

theorem accept_eq_some_iff {n : ℕ} (a : Fin n) (x : Fin (bucket n)) :
    accept n x = some a ↔ x = embed a := by
  unfold accept
  split_ifs with h
  · simp only [Option.some.injEq]
    constructor <;> intro he <;> apply Fin.ext
    · exact congrArg Fin.val he
    · exact congrArg Fin.val he
  · constructor
    · simp
    · intro he
      have hh := congrArg Fin.val he
      exact (h (by simpa only [hh, embed] using a.isLt)).elim

noncomputable section

/-- The law is the literal independent fair-bit recursion used by `Cells`. -/
def bitsLaw (b : ℕ) : PMF (Bits b) := FiniteGridSampler.cellsPMF 2 b

theorem word_uniform (b : ℕ) :
    (bitsLaw b).map (decode b) = PMF.uniformOfFintype (Fin (2 ^ b)) := by
  rw [bitsLaw, FiniteGridSampler.cellsPMF_uniform]
  exact uniform_map_equiv (decode b)

def trial (n : ℕ) : PMF (Option (Fin n)) :=
  ((bitsLaw (width n)).map (decode (width n))).map (accept n)

theorem trial_eq (n : ℕ) : trial n =
    (PMF.uniformOfFintype (Fin (bucket n))).map (accept n) := by
  rw [trial, word_uniform]
  rfl

theorem trial_some {n : ℕ} (a : Fin n) :
    trial n (some a) = (bucket n : ℝ≥0∞)⁻¹ := by
  rw [trial_eq, PMF.map_apply, tsum_fintype]
  have h (x : Fin (bucket n)) : (some a = accept n x) ↔ x = embed a := by
    rw [eq_comm]
    exact accept_eq_some_iff a x
  simp_rw [h, PMF.uniformOfFintype_apply, Fintype.card_fin]
  simp

theorem trial_some_toReal {n : ℕ} (a : Fin n) :
    (trial n (some a)).toReal = (bucket n : ℝ)⁻¹ := by
  simp [trial_some]

theorem trial_none_toReal (n : ℕ) :
    (trial n none).toReal = 1 - (n : ℝ) / bucket n := by
  have h := sum_probability_toReal (trial n)
  rw [Fintype.sum_option] at h
  simp only [trial_some_toReal, Finset.sum_const, Finset.card_univ,
    Fintype.card_fin, nsmul_eq_mul, ← div_eq_mul_inv] at h
  linarith

/-- Positive bounds accept at least half the words, including the bound one. -/
theorem trial_reject_le_half {n : ℕ} (hn : 0 < n) :
    (trial n none).toReal ≤ (1 : ℝ) / 2 := by
  rw [trial_none_toReal]
  have hp : (0 : ℝ) < bucket n := by exact_mod_cast bucket_pos n
  have hbound : (bucket n : ℝ) ≤ 2 * (n : ℝ) := by exact_mod_cast bucket_le_twice hn
  have hh : (1 : ℝ) / 2 ≤ (n : ℝ) / bucket n :=
    (le_div_iff₀ hp).mpr (by linarith)
  linarith

end
end DirectedFlowCutGap.FairBitWords
